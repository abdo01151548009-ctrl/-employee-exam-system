const shouldBelong = checked.includes(e.id);
      if(shouldBelong && e.teamLeaderId!==ldId){
        await setDoc('employees', e.id, { teamLeaderId: ldId });
      } else if(!shouldBelong && e.teamLeaderId===ldId){
        await setDoc('employees', e.id, { teamLeaderId: null });
      }
    }
    alert('تم حفظ أعضاء الفريق بنجاح!');
    state.view='employees-teams'; routeRender();
  });

  document.querySelectorAll('[data-delemp]').forEach(b=>b.onclick=async()=>{
    if(!confirm('هل أنت متأكد من حذف هذا الموظف؟')) return;
    await delDoc('employees', b.dataset.delemp);
    state.view='employees-teams'; routeRender();
  });

  document.getElementById('eSearch').oninput=async (e)=>{
    const emps=await listAll('employees');
    const leaders=await listAll('teamLeaders');
    const term=e.target.value.toLowerCase();
    const filtered=emps.filter(emp=>(emp.name||'').toLowerCase().includes(term) || (emp.dept||'').toLowerCase().includes(term) || (emp.empId||'').toLowerCase().includes(term));
    document.getElementById('empList').innerHTML=renderEmpList(filtered, leaders);
    attachSelectAll('empSelectAll','empChk');
  };
  document.getElementById('empBulkDelete').onclick=()=>bulkDeleteSelected('employees','empChk');
  attachSelectAll('empSelectAll','empChk');
